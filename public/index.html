if (type === 'shoot') { // Shotgun: crack (väldigt kort) + blast (bandkänsla) + thump (pitch-drop) + softclip const sr = audioCtx.sampleRate; const dur = 0.22; const buf = audioCtx.createBuffer(1, Math.floor(sr * dur), sr); const d = buf.getChannelData(0); // Parametrar (små ändringar här gör stor skillnad) const drive = 3.2; // mer = mer "punch" const crackHz = 2600; // 2–4 kHz brukar funka const blastDecay = 70; // kort blast const bodyDecay = 22; // lite längre body const crackDecay = 1400; // superkort transient // Enkel lowpass-state för att få "boom" ur brus utan filter-noder let lp = 0; const lpAlpha = 0.10; // lägre = mörkare/boomigare blast // För pitch-drop behöver vi fas-ackumulation (sin(2π f(t) t) blir lätt "fel" när f ändras) let phase = 0; for (let i = 0; i < d.length; i++) { const t = i / sr; // Envelopes const envBlast = Math.exp(-t * blastDecay); const envBody = Math.exp(-t * bodyDecay); const envCrack = Math.exp(-t * crackDecay); // 1) Crack: kort bright komponent + lite brus const crack = (Math.sin(2 * Math.PI * crackHz * t) * 0.65 + (Math.random() * 2 - 1) * 0.35) * envCrack; // 2) Blast: brus som "formas" (mörk boom + lite bite) const n = (Math.random() * 2 - 1); lp = lp + lpAlpha * (n - lp); // lågpassad komponent (boom) const blast = (n * 0.55 + lp * 0.85) * envBlast; // 3) Thump/body: pitch drop 130 -> 70 Hz, med fas-ackumulation const f = 70 + (130 - 70) * Math.exp(-t * 45); phase += (2 * Math.PI * f) / sr; const body = Math.sin(phase) * envBody; // Mix: shotgun = mer blast+body än pistol let s = crack * 0.95 + blast * 1.10 + body * 0.80; // Soft clip (saturation) s = Math.tanh(s * drive); // Slutlig nivå d[i] = s * 0.9; } // OBS: Jag antar att du efter detta använder din befintliga uppspelningskod (source/gain/start) }
